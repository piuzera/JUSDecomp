// No public constructor combines arbitrary CPU and ISA fields.
#[derive(Clone, Debug)]
enum Architecture { Arm7TdmiV4T, Arm946EV5Te }
impl Architecture {
    fn parse_flags(&self, ual: bool) -> ParseFlags { unimplemented!("fixed explicit version; never Default") }
}

// ProgramIdentity already has Eq, but not Ord/Hash. Keep address indexes local;
// do not pretend it can be dropped into the existing BTreeMap key unchanged.
#[derive(Clone, Debug, PartialEq, Eq)]
enum ProgramKey {
    Verified(ProgramIdentity),
    LegacyArm9(Arm9Scope),
}
#[derive(Clone, Debug)]
pub struct Arm9Scope { token: Arc<OpaqueScopeToken> }
#[derive(Debug)]
struct OpaqueScopeToken { /* private unique instance identity, no fabricated hashes */ }
impl PartialEq for Arm9Scope {
    fn eq(&self, other: &Self) -> bool { unimplemented!("Arc::ptr_eq on allocated scope tokens") }
}
impl Eq for Arm9Scope {}
impl Arm9Scope {
    pub fn new() -> Self { unimplemented!("one allocation per legacy Program load; clones retain scope") }
}
#[derive(Clone, Debug, PartialEq, Eq)]
enum RegionKey { Arm7(Arm7RegionKind), Arm9(ModuleKind) }
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct AnalysisIdentity { program: ProgramKey, cpu: Cpu, region: RegionKey }

#[derive(Clone, Debug)]
pub struct AnalysisContext {
    identity: AnalysisIdentity,
    architecture: Architecture,
    destinations: Arc<AddressContext>,
    semantics: AnalyzerSemantics,
}
#[derive(Clone, Debug)]
enum AnalyzerSemantics { Arm9Legacy, ExplicitInstructionSpan }
impl AnalysisContext {
    pub fn arm9_legacy(identity: &AnalysisIdentity) -> Result<Self, AnalysisInputError> { unimplemented!("reject non-ARM9 identity; old heuristics/address rules") }
    pub fn isa(&self) -> InstructionSet { unimplemented!() }
    pub fn identity(&self) -> &AnalysisIdentity { unimplemented!() }
    pub(crate) fn flags(&self, ual: bool) -> ParseFlags { unimplemented!() }
}

// Each map belongs to exactly one program and CPU. Immutable owned declarations.
// Checked physical ranges establish MappingKind only. Code declarations stay separate.
#[derive(Clone, Debug)]
pub struct AddressContext {
    program: ProgramKey, cpu: Cpu,
    mappings: Vec<PhysicalMapping>, declarations: Vec<ExecutableDeclaration>,
}
#[derive(Clone, Debug)]
struct PhysicalMapping { region: RegionKey, initialized: Range<u32>, bss: Range<u32> }
#[derive(Clone, Debug)]
pub struct ExecutableDeclaration {
    // Construct through checked declaration constructor, not public fields.
    region: RegionKey, span: Range<u32>, mode: InstructionMode,
    evidence: DeclarationEvidence,
}
#[derive(Clone, Debug)]
struct DeclarationEvidence { /* caller-supplied evidence reference retained in report */ }
impl AddressContext {
    pub fn arm7_physical(view: &Arm7View<'_>, executable: &[ExecutableDeclaration])
        -> Result<Self, AnalysisInputError> { unimplemented!("reject mismatched identities, overlapping contradictory mode declarations") }
    fn classify(&self, reference: TargetReference) -> TargetResolution { unimplemented!("mapping, execution and mode are independent") }
}

// Source borrows immutable bytes, while its owned identity may be stored in Function.
// The ARM7 bridge copies identity() accessors, initialized(), bss(), bytes().
// It never accepts the mutable descriptive TargetPolicy as decoder authority.
pub struct AnalysisSource<'a> { identity: AnalysisIdentity, bytes: &'a [u8], initialized: Range<u32> }
impl<'a> AnalysisSource<'a> {
    pub fn arm7(input: &'a Arm7ModuleInput<'_>) -> Self {
        unimplemented!("copy only checked input identity and ranges; borrow immutable bytes")
    }
    pub fn arm9_config(scope: &Arm9Scope, kind: ModuleKind, base: u32, bytes: &'a [u8])
        -> Result<Self, AnalysisInputError> { unimplemented!("checked base+length; explicit legacy ARM9 scope") }
    pub fn identity(&self) -> &AnalysisIdentity { unimplemented!() }
}

// A strict ARM7 instruction span has complete V4T decoding for all requested bytes.
// This type is not a function extent or a declaration that its bytes execute.
pub struct AnalysisSpan<'a> {
    source: AnalysisSource<'a>, context: AnalysisContext,
    mode: InstructionMode, requested: Range<u32>,
    instructions: Vec<DecodedInstruction>,
}
impl<'a> AnalysisSpan<'a> {
    pub fn arm7(input: &'a Arm7ModuleInput<'_>, mode: InstructionMode, span: Range<u32>,
        addresses: &AddressContext) -> Result<Self, AnalysisInputError> {
        unimplemented!("check full context identity; call existing decode_span(mode, span); own context copy; retain exact span")
    }
}
pub struct DecodedInstruction { address: u32, byte_len: u8, ins: Ins, parsed: ParsedIns }
pub enum AnalysisInputError {
    IdentityMismatch, ContextMismatch, Bounds, Overflow, Alignment,
    EmptyOrReversed, TruncatedInstruction { address: u32 }, IllegalInstruction { address: u32 },
    ContradictoryDeclaration,
}

pub struct SpanOptions { candidate: Option<CandidateRequest>, check_defs_uses: bool }
impl SpanOptions {
    pub fn observations() -> Self { unimplemented!("no function claim; optional semantic facts only") }
    pub fn candidate(name: String, entry: u32) -> Self { unimplemented!("caller candidate, not a proven boundary") }
}
struct CandidateRequest { name: String, entry: u32 }
pub enum Completion {
    InstructionSpanOnly,
    CandidateIncomplete { reason: StopReason },
    CandidateRejected { reason: StopReason },
    CandidateTerminated { derived_extent: Range<u32> },
}
pub enum StopReason { DecodeLimit, UnresolvedControlFlow, InvalidEntry, UnsupportedHeuristic, OutsideSpan }
pub struct SpanAnalysis {
    context: AnalysisContext, mode: InstructionMode, requested: Range<u32>,
    instructions: Vec<DecodedInstruction>, targets: Vec<TargetResolution>,
    completion: Completion, function: Option<Function>,
}
impl SpanAnalysis {
    pub fn context(&self) -> &AnalysisContext { unimplemented!() }
    pub fn completion(&self) -> &Completion { unimplemented!() }
    pub fn function(&self) -> Option<&Function> { unimplemented!("Some only for actual bounded termination; no known_end override") }
    pub fn write_json<W: io::Write>(&self, w: &mut W) -> io::Result<()> {
        unimplemented!("private serde DTO; full ARM7 ProgramIdentity and scope included")
    }
}

struct TargetReference { source: u32, address: Option<u32>, kind: TargetKind, mode: TargetMode }
enum TargetKind { B, Bl, Blx, Bx, JumpTable }
enum TargetMode { Arm, Thumb, Unknown }
pub enum TargetResolution {
    DeclaredExecutable { address: u32, region: RegionKey, mode: InstructionMode },
    Unresolved { address: Option<u32>, mapping: MappingKnowledge, mode: TargetMode, reason: UnresolvedReason },
}
pub enum MappingKnowledge { Initialized(RegionKey), Bss(RegionKey), OutsideDeclaredMappings, Unknown }
pub enum UnresolvedReason { ExecutionUndeclared, ModeUnknown, BytesUnavailable, RegisterValueUnknown, ConflictingDeclaration }

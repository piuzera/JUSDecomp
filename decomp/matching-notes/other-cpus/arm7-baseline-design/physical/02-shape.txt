// Verify tool/version pins; reject nonempty/symlink output. Derive a private NativeLayout
// solely from checked view/envelope. Write fixed-name original region/table files and
// four opaque assembly objects. Invoke pinned clang/lld via Command arguments, no shell.
// Hash actual selected files/commands/script/map/ELF. Validate object and final ELF
// sections/attributes/PHDRs/entry. Reconstruct by checked stored offsets, not VMA order.
// Compare all linked bytes to view bytes and checked hashes/params/table; rehash artifacts
// and native tools, then privately construct VerifiedOpaqueBaseline. No partial success.

Internal NativeLayout:
    identity: borrowed ProgramIdentity
    entry: checked header entry
    regions: ordered Vec<NativeRegion>
    table: separate TableLayout
    stored_bytes: checked image size

Internal NativeRegion:
    kind: Startup | Autoload(original_index)
    stored_extent: checked image-relative Range<u32>
    vma: checked initialized runtime range
    lma: checked base + stored extent, checked arithmetic
    bss: checked runtime range
    initialized_bytes: borrowed original bytes

Internal Elf32PhysicalView<'a>:              // bounded reader, not general ELF toolkit
    // Checked ELF32LE header, section headers/string table, program headers,
    // fixed symbol table and ARM attributes.
    // Checked arithmetic/slices for all counts/offsets; reject extended numbering,
    // unsupported machine/type/entry/section/segment forms and duplicate names.
    // Expose only our expected PROGBITS/NOBITS bytes, actual ranges and attributes.
    // Never treat a NOBITS section as a file byte slice.

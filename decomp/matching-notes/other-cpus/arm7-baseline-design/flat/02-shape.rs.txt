// Extend the checked-view ownership boundary with retained physical authority.
// Private fields are constructed only after the existing actual-byte checks.
// Existing checked() callers can retain this envelope without a sidecar pin;
// the baseline's PinnedArm7Input separately requires consumed sidecar bytes.
pub struct CheckedArm7ImageEnvelope<'rom> {
    identity: ProgramIdentity,
    header: CheckedHeaderIdentity,
    image: &'rom [u8],
    image_offset_in_program: u32,
    original_entry: u32,
    params_extent_in_image: Range<u32>,
    params_sha256: Sha256,
    regions: Vec<CheckedPhysicalRegion>,
    table_extent_in_image: Range<u32>,
    table_sha256: Sha256,
}

// These descriptions contain physical offsets/ranges only. No Code/Data kind.
pub struct CheckedPhysicalRegion {
    kind: Arm7RegionKind,
    stored_extent: Range<u32>,
    initialized_runtime: Range<u32>,
    bss_runtime: Range<u32>,
    sha256: Sha256,
}

// The exact consumed pin cannot be installed by an arbitrary caller.
pub struct PinnedArm7Input<'rom> {
    view: Arm7View<'rom>,
    consumed_sidecar: ConsumedSidecarPin,
}
struct ConsumedSidecarPin {
    sha256: Sha256,
    size_bytes: u64,
}
impl<'rom> PinnedArm7Input<'rom> {
    pub fn load(
        parent: &'rom [u8],
        selected_program: ProgramSelector,
        sidecar_bytes: &[u8],
        expected_sidecar_sha256: Sha256,
    ) -> Result<Self, BaselineInputError> {
        // Hash consumed sidecar bytes before parsing. Select exactly one layout
        // by complete program selector; reject duplicate/missing matches.
        // Invoke existing Arm7Layout::checked on actual parent bytes. Keep the
        // envelope built there; never reconstruct it from later mutable layout.
        // Retain sidecar digest and original program identity. No raw wire types
        // or mutable public constructors escape this boundary.
        unimplemented!()
    }
    pub fn view(&self) -> &Arm7View<'rom> { unimplemented!() }
}

pub struct FlatArm7Baseline;
impl FlatArm7Baseline {
    pub fn build(
        input: &PinnedArm7Input<'_>,
        tools: &NativeTools,
        fresh_private_output: &Path,
    ) -> Result<OpaqueStorageBaseline, BaselineFailure> {
        // Claim fresh output; bind source/tool/input hashes before work.
        // Materialize checked image as private native-link input, preserving
        // startup+autoload payloads+the exact original loader table.
        // Make a data-only ARM relocatable input with LLVM objcopy. Require
        // .arm7_stored PROGBITS, flags0, offset coordinates, alignment4.
        // Produce ARMv4T attributes using clang; reject unexpected payloads.
        // LLVM's empty default executable .text may be removed ONLY after
        // proving zero length; preserve attributes and capture raw/clean hashes.
        // Run native lld with explicit script and entry0. No ARM9 normalizer,
        // no LCF translator, no executable symbols or mapping-symbol invention.
        // Parse output ELF, enforce exact structural contract, reconstruct only
        // from .arm7_stored, and compare every byte to checked original image.
        // Independently verify image params/table/regions/BSS against retained
        // physical envelope; ELF does not claim those runtime placements.
        // Final freshness checks bind original inputs, actual outputs and tools.
        // Record every stage and emit success receipt only after all gates pass.
        unimplemented!()
    }
}

from dataclasses import dataclass
from pathlib import Path
from typing import Mapping

@dataclass(frozen=True)
class CurrentBuild:
    directory: Path
    build_id: str
    started_ns: int

@dataclass(frozen=True)
class ChildApprovals:
    # Private approval loader verifies role-specific pins and complete source context.
    analyzer_approval: Path
    codec_approval: Path
    native_tools_approval: Path

@dataclass(frozen=True)
class ProgramIdentity:
    parent_sha256: str
    nitrofs_path: str | None
    file_id: int | None
    start: int
    end: int
    original_sha256: str

@dataclass(frozen=True)
class ExecutableWrite:
    identity: ProgramIdentity
    cpu: str
    rom_offset: int
    data: bytes
    expected_sha256: str
    source_file: Path
    source_file_sha256: str
    source_slice_offset: int
    linked_elf: Path
    linked_elf_sha256: str
    source_bytes: int = 0

@dataclass(frozen=True)
class VerifiedProgramWrites:
    # Private constructor; built only by a successful live recheck.
    writes: tuple[ExecutableWrite, ExecutableWrite, ExecutableWrite]
    child_identity: ProgramIdentity

        raise NotImplementedError

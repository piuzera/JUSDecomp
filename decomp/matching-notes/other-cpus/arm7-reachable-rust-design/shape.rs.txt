/// A caller assumption tied to an existing checked observation.
/// Mode and full program/CPU/region identity are inherited from observation.
/// connect validates address as an instruction start and records assumption.
/// The header can justify an address; it does not certify mode or execution.
#[derive(Clone, Copy)]
pub struct RootRequest<'o, 'm: 'o> {
    pub observation: &'o SpanObservation<'m>,
    pub address: u32,
    pub assumption: &'o str,
}

/// Why an original transfer could not join a selected instruction interpretation.
/// This classifies supplied evidence, never actual execution or function ends.
#[derive(Debug, PartialEq, Eq)]
pub enum FrontierReason {
    OutsideSelection,
    InstructionInterior,
    ModeConflict,
    Bss,
    Unmapped,
    Indirect,
}

#[derive(Debug)]
pub enum ConnectError {
    EmptyObservations,
    EmptyRoots,
    ForeignProgram,
    WrongTargetPolicy,
    DuplicateOrOverlappingSameModeSelection,
    ForeignRoot,
    RootNotInstructionStart,
    EmptyRootAssumption,
    DuplicateRoot,
    InconsistentMappedIdentity,
    CapacityOverflow,
}

/// Owns graph structure; borrows immutable checked evidence.
/// No public fields, Deserialize, raw-JSON constructor, or mutation methods.
/// Serde report serialization is a read-only custom implementation.
pub struct CandidateGraph<'o, 'm: 'o> {
    observations: &'o [SpanObservation<'m>],
    nodes: Vec<SelectedInstruction<'o, 'm>>,
    edges: Vec<SelectedEdge<'o, 'm>>,
    roots: Vec<RootTraversal<'o, 'm>>,
}

// Private construction data. Indexes never become caller-controlled graph handles.
struct SelectedInstruction<'o, 'm: 'o> {
    observation: &'o SpanObservation<'m>,
    instruction: &'o InstructionObservation,
    outgoing: Range<usize>,
    mode_conflict: bool,
}
struct SelectedEdge<'o, 'm: 'o> {
    source_node: usize,
    original: &'o TransferObservation<'m>,
    destination: Destination,
}
enum Destination {
    SelectedNode(usize),
    Frontier(FrontierReason),
}
struct RootTraversal<'o, 'm: 'o> {
    request: RootRequest<'o, 'm>,
    visits: Vec<Visit>,
    frontier_sites: Vec<FrontierSite>,
}
enum Visit {
    Unvisited,
    Root,
    ViaEdge(usize),
}
enum FrontierSite {
    Root(FrontierReason),
    Edge(usize),
}
